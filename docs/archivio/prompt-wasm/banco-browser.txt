#[no_mangle]
pub unsafe extern "C" fn inflate_many(src: *const u8, lens: *const u32, n: usize, dst: *mut u8, cap: usize) -> usize {
    let mut off = 0usize; let mut out = 0usize;
    for i in 0..n {
        let l = *lens.add(i) as usize;
        let s = std::slice::from_raw_parts(src.add(off), l);
        let d = std::slice::from_raw_parts_mut(dst.add(out), cap - out);
        let mut z = Decompress::new(true);
        z.decompress(s, d, FlushDecompress::Finish).unwrap();
        out += z.total_out() as usize; off += l;
    }
    out
}
